cmake_minimum_required(VERSION 3.16)
project(OscillatorNetwork CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT CMAKE_BUILD_TYPE)
  set(CMAKE_BUILD_TYPE Release)
endif()
option(OSCNET_SANITIZE "Build with AddressSanitizer and UBSan" OFF)
if(OSCNET_SANITIZE)
  add_compile_options(-fsanitize=address,undefined -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(NOT MSVC)
  add_compile_options(-Wall -Wextra)
endif()

add_executable(oscnet src/main.cpp)
target_include_directories(oscnet PRIVATE include)

enable_testing()
find_path(DOCTEST_INCLUDE doctest/doctest.h)
if(DOCTEST_INCLUDE)
  add_executable(oscnet_tests tests/test_oscnet.cpp)
  target_include_directories(oscnet_tests PRIVATE include ${DOCTEST_INCLUDE})
  add_test(NAME oscnet_tests COMMAND oscnet_tests)
endif()
